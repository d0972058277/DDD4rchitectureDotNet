# 訊息佇列關聯性識別碼使用範例

本文檔展示如何在 RabbitMQ 和 Kafka 中使用關聯性識別碼功能，實現跨服務的分散式追蹤。

## 📦 初始設定

首先，在服務中加入訊息佇列關聯性支援：

```csharp
services.AddCorrelation(options =>
{
    options.RequestHeader = "X-Correlation-ID";
    options.AddToLoggingScope = true;
})
.AddMessagingCorrelation();
```

## 🐰 RabbitMQ 整合

### 發布者 (發布訊息)

```csharp
public class OrderEventPublisher
{
    private readonly IConnection _connection;
    private readonly RabbitMqCorrelationIntegration _correlationIntegration;
    private readonly ILogger<OrderEventPublisher> _logger;

    public OrderEventPublisher(
        IConnection connection, 
        RabbitMqCorrelationIntegration correlationIntegration,
        ILogger<OrderEventPublisher> logger)
    {
        _connection = connection;
        _correlationIntegration = correlationIntegration;
        _logger = logger;
    }

    public async Task PublishOrderCreatedAsync(OrderCreatedEvent orderEvent)
    {
        using var channel = _connection.CreateModel();
        
        // 建立訊息屬性
        var properties = channel.CreateBasicProperties();
        properties.Persistent = true;
        properties.MessageId = Guid.NewGuid().ToString();
        properties.Timestamp = new AmqpTimestamp(DateTimeOffset.UtcNow.ToUnixTimeSeconds());
        
        // 注入關聯性 ID 到訊息屬性
        _correlationIntegration.InjectCorrelationId(properties);
        
        var body = JsonSerializer.SerializeToUtf8Bytes(orderEvent);
        
        _logger.LogInformation("發布訂單建立事件: {OrderId}", orderEvent.OrderId);
        
        channel.BasicPublish(
            exchange: "orders",
            routingKey: "order.created",
            basicProperties: properties,
            body: body);
    }

    public async Task PublishMultipleEventsAsync(IEnumerable<OrderEvent> events)
    {
        using var channel = _connection.CreateModel();
        
        foreach (var orderEvent in events)
        {
            var properties = channel.CreateBasicProperties();
            
            // 每個訊息都會自動注入當前的關聯性 ID
            _correlationIntegration.InjectCorrelationId(properties);
            
            var body = JsonSerializer.SerializeToUtf8Bytes(orderEvent);
            
            channel.BasicPublish("orders", orderEvent.RoutingKey, properties, body);
        }
    }
}
```

### 消費者 (處理訊息)

```csharp
public class OrderEventConsumer
{
    private readonly RabbitMqCorrelationIntegration _correlationIntegration;
    private readonly ICorrelationContextFactory _correlationFactory;
    private readonly IOrderProcessor _orderProcessor;
    private readonly ILogger<OrderEventConsumer> _logger;

    public OrderEventConsumer(
        RabbitMqCorrelationIntegration correlationIntegration,
        ICorrelationContextFactory correlationFactory,
        IOrderProcessor orderProcessor,
        ILogger<OrderEventConsumer> logger)
    {
        _correlationIntegration = correlationIntegration;
        _correlationFactory = correlationFactory;
        _orderProcessor = orderProcessor;
        _logger = logger;
    }

    public async Task HandleOrderCreatedAsync(BasicDeliverEventArgs args)
    {
        // 提取並建立關聯性上下文
        using var correlationContext = _correlationIntegration.CreateCorrelationContext(
            args.BasicProperties, _correlationFactory);
        
        try
        {
            var orderEvent = JsonSerializer.Deserialize<OrderCreatedEvent>(args.Body.Span);
            
            // 在日誌中自動包含關聯性 ID
            _logger.LogInformation("處理訂單建立事件: {OrderId}", orderEvent.OrderId);
            
            // 在關聯性上下文中處理訂單
            await _orderProcessor.ProcessOrderAsync(orderEvent);
            
            _logger.LogInformation("訂單處理完成: {OrderId}", orderEvent.OrderId);
        }
        catch (Exception ex)
        {
            // 錯誤日誌中也會包含關聯性 ID
            _logger.LogError(ex, "處理訂單事件時發生錯誤");
            throw;
        }
    }

    public async Task HandleBatchMessagesAsync(IEnumerable<BasicDeliverEventArgs> messages)
    {
        foreach (var message in messages)
        {
            // 每個訊息都有獨立的關聯性上下文
            using var correlationContext = _correlationIntegration.CreateCorrelationContext(
                message.BasicProperties, _correlationFactory);
            
            await ProcessSingleMessageAsync(message);
        }
    }
}
```

## ⚡ Kafka 整合

### 發布者 (發布訊息)

```csharp
public class PaymentEventProducer
{
    private readonly IProducer<string, PaymentProcessedEvent> _producer;
    private readonly KafkaCorrelationIntegration _correlationIntegration;
    private readonly ILogger<PaymentEventProducer> _logger;

    public PaymentEventProducer(
        IProducer<string, PaymentProcessedEvent> producer,
        KafkaCorrelationIntegration correlationIntegration,
        ILogger<PaymentEventProducer> logger)
    {
        _producer = producer;
        _correlationIntegration = correlationIntegration;
        _logger = logger;
    }

    public async Task PublishPaymentProcessedAsync(PaymentProcessedEvent paymentEvent)
    {
        var message = new Message<string, PaymentProcessedEvent>
        {
            Key = paymentEvent.PaymentId,
            Value = paymentEvent,
            Timestamp = Timestamp.Default
        };
        
        // 注入關聯性 ID 到訊息標頭
        _correlationIntegration.InjectCorrelationId(message);
        
        _logger.LogInformation("發布付款處理事件: {PaymentId}", paymentEvent.PaymentId);
        
        var deliveryResult = await _producer.ProduceAsync("payments", message);
        
        _logger.LogDebug("訊息已發送到分區 {Partition}, 偏移量 {Offset}", 
            deliveryResult.Partition, deliveryResult.Offset);
    }

    public async Task PublishWithCustomHeadersAsync(PaymentEvent paymentEvent)
    {
        var headers = new Headers
        {
            { "event-type", Encoding.UTF8.GetBytes(paymentEvent.GetType().Name) },
            { "source-service", Encoding.UTF8.GetBytes("payment-service") }
        };
        
        // 直接對 Headers 注入關聯性 ID
        _correlationIntegration.InjectCorrelationId(headers);
        
        var message = new Message<string, PaymentEvent>
        {
            Key = paymentEvent.PaymentId,
            Value = paymentEvent,
            Headers = headers
        };
        
        await _producer.ProduceAsync("payment-events", message);
    }
}
```

### 消費者 (處理訊息)

```csharp
public class PaymentEventConsumer
{
    private readonly KafkaCorrelationIntegration _correlationIntegration;
    private readonly ICorrelationContextFactory _correlationFactory;
    private readonly IPaymentProcessor _paymentProcessor;
    private readonly ILogger<PaymentEventConsumer> _logger;

    public PaymentEventConsumer(
        KafkaCorrelationIntegration correlationIntegration,
        ICorrelationContextFactory correlationFactory,
        IPaymentProcessor paymentProcessor,
        ILogger<PaymentEventConsumer> logger)
    {
        _correlationIntegration = correlationIntegration;
        _correlationFactory = correlationFactory;
        _paymentProcessor = paymentProcessor;
        _logger = logger;
    }

    // 從完整訊息中提取關聯性上下文
    public async Task HandlePaymentMessageAsync(ConsumeResult<string, PaymentProcessedEvent> consumeResult)
    {
        // 從訊息中提取並建立關聯性上下文
        using var correlationContext = _correlationIntegration.CreateCorrelationContext(
            consumeResult.Message, _correlationFactory);
        
        var paymentEvent = consumeResult.Message.Value;
        
        _logger.LogInformation("處理付款事件: {PaymentId}, 分區: {Partition}, 偏移量: {Offset}", 
            paymentEvent.PaymentId, consumeResult.Partition, consumeResult.Offset);
        
        await _paymentProcessor.ProcessPaymentAsync(paymentEvent);
    }
    
    // 僅從標頭中提取關聯性上下文
    public async Task HandleFromHeadersOnlyAsync(ConsumeResult<string, PaymentProcessedEvent> consumeResult)
    {
        using var correlationContext = _correlationIntegration.CreateCorrelationContext(
            consumeResult.Message.Headers, _correlationFactory);
        
        // 處理訊息...
        await ProcessPaymentAsync(consumeResult.Message.Value);
    }

    // 手動關聯性 ID 處理
    public async Task HandleWithManualExtractionAsync(ConsumeResult<string, PaymentProcessedEvent> consumeResult)
    {
        // 手動提取關聯性 ID
        var correlationId = _correlationIntegration.ExtractCorrelationId(consumeResult.Message);
        
        if (string.IsNullOrEmpty(correlationId))
        {
            _logger.LogWarning("收到沒有關聯性 ID 的 Kafka 訊息");
            correlationId = Guid.NewGuid().ToString(); // 生成新的 ID
        }

        using var context = _correlationFactory.Create(correlationId, "kafka-header");
        
        await ProcessPaymentAsync(consumeResult.Message.Value);
    }
}
```

### Kafka 批次處理

```csharp
public class KafkaBatchProcessor
{
    private readonly KafkaCorrelationIntegration _correlationIntegration;
    private readonly ICorrelationContextFactory _correlationFactory;

    public async Task ProcessBatchAsync(IEnumerable<ConsumeResult<string, OrderEvent>> batch)
    {
        var tasks = batch.Select(async consumeResult =>
        {
            // 每個訊息都有獨立的關聯性上下文
            using var correlationContext = _correlationIntegration.CreateCorrelationContext(
                consumeResult.Message, _correlationFactory);
            
            await ProcessSingleOrderAsync(consumeResult.Message.Value);
        });

        await Task.WhenAll(tasks);
    }

    public async Task ProcessWithGroupingAsync(IEnumerable<ConsumeResult<string, OrderEvent>> batch)
    {
        // 按關聯性 ID 分組處理
        var groups = batch.GroupBy(r => 
            _correlationIntegration.ExtractCorrelationId(r.Message) ?? "unknown");

        foreach (var group in groups)
        {
            using var correlationContext = string.IsNullOrEmpty(group.Key) || group.Key == "unknown"
                ? null 
                : _correlationFactory.Create(group.Key, "kafka-header");

            await ProcessOrderGroupAsync(group.Select(r => r.Message.Value));
        }
    }
}
```

## 🔧 直接協議處理器使用

對於進階情境，可以直接使用協議處理器：

```csharp
public class AdvancedMessagingService
{
    private readonly ICorrelationPropagator _propagator;

    public AdvancedMessagingService(ICorrelationPropagator propagator)
    {
        _propagator = propagator;
    }

    public void HandleRabbitMqMessage(IBasicProperties properties)
    {
        // 直接注入/提取
        _propagator.Inject(properties);
        var correlationId = _propagator.Extract(properties);
        
        Console.WriteLine($"RabbitMQ 關聯性 ID: {correlationId}");
    }

    public void HandleKafkaHeaders(Headers headers)
    {
        // 直接注入/提取
        _propagator.Inject(headers);
        var correlationId = _propagator.Extract(headers);
        
        Console.WriteLine($"Kafka 關聯性 ID: {correlationId}");
    }
    
    public void HandleGenericDictionary(IDictionary<string, string> metadata)
    {
        // 適用於任何 IDictionary 實作 (fallback 機制)
        _propagator.Inject(metadata);
        var correlationId = _propagator.Extract(metadata);
        
        Console.WriteLine($"泛型容器關聯性 ID: {correlationId}");
    }

    public void HandleMixedProtocols()
    {
        // 同一個傳播器可以處理多種協議
        var rabbitProperties = new BasicProperties();
        var kafkaHeaders = new Headers();
        var customMetadata = new Dictionary<string, object>();

        // 注入到所有協議
        _propagator.Inject(rabbitProperties);
        _propagator.Inject(kafkaHeaders);
        _propagator.Inject(customMetadata);

        // 從所有協議提取
        var rabbitId = _propagator.Extract(rabbitProperties);
        var kafkaId = _propagator.Extract(kafkaHeaders);
        var customId = _propagator.Extract(customMetadata);

        // 所有 ID 都應該相同
        Debug.Assert(rabbitId == kafkaId && kafkaId == customId);
    }
}
```

```csharp
public class MessageProcessorWithLogging
{

    {
        var correlationId = _correlationService.CorrelationId;
        
        using var scope = _logger.BeginScope(new Dictionary<string, object>
        {
            ["CorrelationId"] = correlationId,
            ["MessageType"] = typeof(T).Name,
            ["ProcessingStartTime"] = DateTime.UtcNow
        });

        _logger.LogInformation("開始處理訊息");
        
        try
        {
            // 處理訊息邏輯
            await ProcessAsync(message);
            
            _logger.LogInformation("訊息處理完成");
        }
        catch (Exception ex)
        {

        }
    }
}
```

### 2. 效能優化

```csharp
public class OptimizedMessageHandler
{
    private static readonly JsonSerializerOptions JsonOptions = new()
    {
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
        DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull
    };

    public async Task ProcessHighThroughputAsync(IEnumerable<ConsumeResult<string, OrderEvent>> batch)
    {
        // 批次提取關聯性 ID，避免重複操作
        var contextTasks = batch.Select(async result =>
        {
            var correlationId = _correlationIntegration.ExtractCorrelationId(result.Message);
            var context = !string.IsNullOrEmpty(correlationId) 
                ? _correlationFactory.Create(correlationId, "kafka-header")
                : null;
            
            return new { Result = result, Context = context };
        });

        var contextsWithResults = await Task.WhenAll(contextTasks);

        try
        {
            await Task.WhenAll(contextsWithResults.Select(async item =>
            {
                using (item.Context)
                {
                    await ProcessSingleOrderAsync(item.Result.Message.Value);
                }
            }));
        }
        finally
        {
            // 確保所有上下文都被正確處置
            foreach (var item in contextsWithResults)
            {
                item.Context?.Dispose();
            }
        }
    }
}
```

## 🔍 疑難排解

### 常見問題與解決方案

**Q: RabbitMQ 訊息中沒有關聯性 ID？**
```csharp
// 檢查是否正確注入
var properties = channel.CreateBasicProperties();
_correlationIntegration.InjectCorrelationId(properties);

// 驗證是否注入成功
var correlationId = _correlationIntegration.ExtractCorrelationId(properties);
Debug.Assert(!string.IsNullOrEmpty(correlationId));
```

**Q: Kafka 標頭中的關聯性 ID 是亂碼？**
```csharp
// 檢查編碼問題
if (headers.TryGetLastBytes("X-Correlation-ID", out var bytes))
{
    var correlationId = Encoding.UTF8.GetString(bytes);
    Console.WriteLine($"提取的關聯性 ID: {correlationId}");
}
```

**Q: 批次處理時關聯性 ID 混亂？**
```csharp
// 確保每個訊息都有獨立的上下文
foreach (var message in batch)
{
    using var correlationContext = CreateCorrelationContext(message);
    // 處理單一訊息
}
```

這個完整的訊息佇列整合方案確保關聯性 ID 能在 RabbitMQ 和 Kafka 等訊息佇列系統中無縫傳播，實現真正的端到端分散式追蹤。
